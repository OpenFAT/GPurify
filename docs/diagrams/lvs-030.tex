% An inverter at x = #1, node prefix #2, nfet text #3, with a capacitor from
% Y to ground whose upper plate is #4 and lower plate #5.
\newcommand{\invcap}[5]{%
  \node[pmos] (#2p) at (#1,2.8) {};
  \node[right=2pt] at (#2p.east) {pfet};
  \draw (#2p.S) -- ++(0,0.3) node[vcc] {VDD};
  \node[nmos] (#2n) at (#1,0.6) {};
  \node[right=2pt,align=left,font=\footnotesize] at (#2n.east) {#3};
  \draw (#2p.D) -- (#2n.D);
  \draw (#2p.D) ++(0,-0.35) coordinate (#2y) -- ++(3.4,0) coordinate (#2c) -- ++(0.6,0) node[right] {Y};
  \draw (#2c) to[C] ++(0,-1.6) coordinate (#2cb) node[ground] {};
  \node[anchor=south west,font=\footnotesize] at ($(#2c)+(0.3,-0.75)$) {#4};
  \node[anchor=north west,font=\footnotesize] at ($(#2c)+(0.3,-0.85)$) {#5};
  \node[anchor=south,font=\footnotesize] at ($(#2c)+(0,0.05)$) {cmim};
  \draw (#2n.S) node[ground] {};
  \draw (#2p.G) -- ++(-0.6,0) coordinate (#2a) -- (#2a |- #2n.G) -- (#2n.G);
  \draw (#2a |- #2p.G) node[left] {A};
}
