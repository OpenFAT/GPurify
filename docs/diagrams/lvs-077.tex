% A two-high nfet stack at x = #1, node prefix #2, gates A over B, each
% transistor labelled #3; (#2i) is its inner net.
\newcommand{\stack}[3]{%
  \node[nmos] (#2t) at (#1,2.7) {};
  \node[nmos] (#2b) at (#1,0.7) {};
  \node[right=1pt,font=\footnotesize] at (#2t.east) {#3};
  \node[right=1pt,font=\footnotesize] at (#2b.east) {#3};
  \draw (#2t.S) -- (#2b.D);
  \coordinate (#2i) at ($(#2t.S)!0.5!(#2b.D)$);
  \draw (#2t.D) -- ++(0,0.3) coordinate (#2y);
  \draw (#2b.S) -- ++(0,-0.3) coordinate (#2s);
  \draw (#2t.G) -- ++(-0.25,0) node[left] {A};
  \draw (#2b.G) -- ++(-0.25,0) node[left] {B};
}
